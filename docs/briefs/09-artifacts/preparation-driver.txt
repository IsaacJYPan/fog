import json, os, subprocess, time
from pathlib import Path
root = Path('/workspace/fog')
art = root / 'docs/briefs/09-artifacts'
records = []
def run(name, args, **kwargs):
    start = time.monotonic()
    with (art / (name + '.txt')).open('a') as log:
        result = subprocess.run(args, stdout=log, stderr=subprocess.STDOUT, **kwargs)
    records.append(dict(name=name, command=args, seconds=time.monotonic()-start,
                        exit_code=result.returncode, cwd=kwargs.get('cwd', str(root)), commit='70cc5eda12345edffe11a998e2c4782f4ed7b161',
                        environment={k:os.environ.get(k) for k in ('OMP_WAIT_POLICY','KMP_BLOCKTIME')}, smoke=True))
    (art / 'preparation-execution.json').write_text(json.dumps(records, indent=2)+'\n')
    if result.returncode: raise SystemExit(result.returncode)
py = str(root / '.venv/bin/python')
run('conversion', [py, '-m', 'fog.scripts.convert_replay_decisions', '--raw', 'data/metamon_gen9ou_r1500', '--out', 'data/metamon_gen9ou_r1500_decisions', '--imputations', '1', '--workers', '4', '--resume'])
run('split', [py, '-m', 'fog.scripts.split_replay_manifest', '--manifest', 'data/metamon_gen9ou_r1500_decisions/manifest.parquet', '--out', 'runs/blueprint/split', '--holdout-ids', 'data/replay_splits/p2_holdout_2000_ids.txt', '--train-sizes', '1000'])
