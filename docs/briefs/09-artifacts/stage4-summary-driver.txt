# Resample whole matched blocks; keep the same bootstrap indices for paired policies.
rng=np.random.default_rng(0)
indices=rng.integers(0,100,size=(10000,100))
play_intervals={'smoke':True,'source':source,'samples':10000,'seed':0,'unit':'matched four-game block','analysis_command':'python runs/order_blind_model/finish_stage4.py','analysis_source':'09-artifacts/stage4-summary-driver.txt','unresolved':'excluded from each resampled denominator; never counted as draws','rows':{},'paired_R4_minus_R1':{}}
resampled={}
for row in play['rows']:
    scores=np.array([np.nan if x is None else x for x in row['game_scores']]).reshape(100,4)
    numerator=np.nansum(scores,axis=1); denominator=np.isfinite(scores).sum(axis=1)
    samples=numerator[indices].sum(axis=1)/denominator[indices].sum(axis=1)
    resampled[row['label']]=samples
    play_intervals['rows'][row['label']]={'score':row['a_score'],'interval95':np.quantile(samples,[.025,.975]).tolist()}
for opponent in ('simpleheuristics','maxdamage'):
    a='R1 vs '+opponent; b='R4 vs '+opponent
    play_intervals['paired_R4_minus_R1'][opponent]={'difference':play_intervals['rows'][b]['score']-play_intervals['rows'][a]['score'],'interval95':np.quantile(resampled[b]-resampled[a],[.025,.975]).tolist()}
(art/'play-block-intervals.json').write_text(json.dumps(play_intervals,indent=2)+'\n')

        b=diags['R4-seed2']['policy']['policy'][group][metric]
        old=diags['R3']['policy']['policy'][group][metric]
        comparisons[group][metric]={'R3':old,'R4':a,'R4_seed2':b,'R4_minus_R3':a-old,'twice_seed_gap':2*abs(a-b),'exceeds_noise':abs(a-old)>2*abs(a-b)}

        a=rare['models']['R4']['groups'][group][metric]
        b=rare['models']['R4-seed2']['groups'][group][metric]
        old=rare['models']['R3']['groups'][group][metric]
        rare_comparison[group][metric]={'R3':old,'R4':a,'R4_seed2':b,'R4_minus_R3':None if a is None else a-old,'twice_seed_gap':None if a is None else 2*abs(a-b),'exceeds_noise':None if a is None else abs(a-old)>2*abs(a-b)}
summary={'smoke':True,'source':source,'stages_complete':[2,3,4],'stage_pending':5,'standard_groups':comparisons,'rare_groups':rare_comparison,'play_rows':play['rows'],'play_has_second_seed':False,'play_strength_gate':False}

    return f'{x*scale:.{n}f} [{bounds[0]*scale:.{n}f}, {bounds[1]*scale:.{n}f}]'
