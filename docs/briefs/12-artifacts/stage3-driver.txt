"""Resumable CPU comparison: old/new R4 training, paired holdout, then greedy matched play."""
import argparse
import json
import os
from pathlib import Path
import subprocess
import sys
import time

import polars as pl

ROOT=Path.cwd()
OUT=ROOT/'runs/replay-parity'
ART=ROOT/'docs/briefs/12-artifacts'
WRAPPER=ART/'runtime-wrapper.txt'
PYTHON=ROOT/'.venv/bin/python'
ENV={**os.environ,'OMP_NUM_THREADS':'2','MKL_NUM_THREADS':'2','OMP_WAIT_POLICY':'PASSIVE',
     'MALLOC_MMAP_THRESHOLD_':'131072','MALLOC_TRIM_THRESHOLD_':'131072'}


def run(module, arguments, log):
    command=[str(PYTHON),'-u',str(WRAPPER),module,*arguments]
    print('Running:', ' '.join(command), flush=True)
    with Path(log).open('a') as stream:
        subprocess.run(command, cwd=ROOT, env=ENV, stdout=stream, stderr=subprocess.STDOUT, check=True)


def train(name):
    run('fog.training.bc',['--config','configs/replay_mechanical_cpu.yaml','--set',
        f'bc.manifest=runs/replay-parity/split-{name}/train_1000.parquet',
        f'bc.holdout_manifest=runs/replay-parity/split-{name}/holdout_manifest.parquet',
        'bc.torch_threads=2',f'log.dir=runs/replay-parity/train-{name}','seed=20261201'],
        f'/tmp/parity-train-{name}.txt')


def alive(pid):
    try:
        return Path(f'/proc/{pid}/stat').read_text().split()[2]!='Z'
    except FileNotFoundError:
        return False


if __name__=='__main__':
    parser=argparse.ArgumentParser()
    parser.add_argument('--existing-old-pid',type=int)
    args=parser.parse_args()
    old_final=OUT/'train-old/checkpoints/bc_final.pt'
    while args.existing_old_pid and alive(args.existing_old_pid) and not old_final.exists():
        time.sleep(10)
    if not old_final.exists():
        train('old')
    while not (OUT/'split-new/training-ready-v12.json').exists():
        time.sleep(10)
    new_final=OUT/'train-new/checkpoints/bc_final.pt'
    if not new_final.exists():
        train('new')
    if not (ART/'holdout-comparison.json').exists():
        run('fog.eval.replay_parity_training',[
            '--old-manifest','runs/replay-parity/split-old/holdout_manifest.parquet',
            '--new-manifest','runs/replay-parity/split-new/holdout_manifest.parquet',
            '--old-metadata','runs/replay-parity/old-metadata',
            '--old-checkpoint','runs/replay-parity/train-old/checkpoints/bc_best.pt',
            '--new-checkpoint','runs/replay-parity/train-new/checkpoints/bc_best.pt',
            '--out','docs/briefs/12-artifacts/holdout-comparison.json'], '/tmp/parity-holdout.txt')
    run('fog.eval.showdown_direct_selection',[
        '--pool','data/teams/pool_b','--device','cpu','--config','configs/replay_mechanical_cpu.yaml',
        '--checkpoint','old=runs/replay-parity/train-old/checkpoints/bc_best.pt',
        '--checkpoint','new=runs/replay-parity/train-new/checkpoints/bc_best.pt',
        '--checkpoint-config','old=configs/replay_mechanical_cpu.yaml',
        '--checkpoint-config','new=configs/replay_mechanical_cpu.yaml',
        '--opponent','simpleheuristics','--opponent','maxdamage',
        '--games','400','--seed','0','--max-turns','300','--progress-every','40','--resume',
        '--out','docs/briefs/12-artifacts/play.json'], '/tmp/parity-play.txt')
    for name in ('old','new'):
        (ART/f'train-{name}-metrics.csv').write_text((OUT/f'train-{name}/metrics.csv').read_text())
    (ART/'stage3-complete.json').write_text(json.dumps({'kind':'smoke','complete':True})+'\n')
