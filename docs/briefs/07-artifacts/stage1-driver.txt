import hashlib
import json
import os
import shutil
import subprocess
import time
from pathlib import Path

root = Path('/workspace/fog')
artifacts = root / 'docs/briefs/07-artifacts'
python = str(root / '.venv/bin/python')
records = []
source_commit = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip()

def run(args, output):
    command = [python, '-m', *args]
    started = time.monotonic()
    record = {'command': command, 'cwd': str(root), 'commit': source_commit, 'smoke': True}
    with open(root / output, 'w') as stream:
        result = subprocess.run(command, cwd=root, stdout=stream, stderr=subprocess.STDOUT)
    record.update(seconds=time.monotonic()-started, exit_code=result.returncode, output=output)
    records.append(record)
    (artifacts / 'stage1-execution.json').write_text(json.dumps(records, indent=2)+'\n')
    if result.returncode:
        raise SystemExit(result.returncode)

while not (root / 'data/metamon_gen9ou_r1500_order_guarded_v3_decisions/quality_report.json').exists():
    time.sleep(5)
run(['fog.scripts.split_replay_manifest', '--manifest',
     'data/metamon_gen9ou_r1500_order_guarded_v3_decisions/manifest.parquet', '--out',
     'runs/blueprint/split', '--holdout-ids',
     'data/replay_splits/p2_holdout_2000_ids.txt', '--train-sizes', '1000'],
    'runs/order_blind/split-R1-guarded-output.txt')
import polars as pl
os.chdir(root)
before = pl.read_parquet('runs/order_blind/leaky_split/holdout_manifest.parquet')
after = pl.read_parquet('runs/blueprint/split/holdout_manifest.parquet')
before_train = pl.read_parquet('runs/order_blind/leaky_split/train_1000.parquet')
after_train = pl.read_parquet('runs/blueprint/split/train_1000.parquet')
ids = set(Path('data/replay_splits/p2_holdout_2000_ids.txt').read_text().splitlines())
assert set(before['id']) == set(after['id']) == ids
assert set(before_train['id']) == set(after_train['id'])
split = {'smoke': True, 'holdout_ids_equal': True, 'holdout_games': len(after),
         'training_ids_equal': True, 'training_games': len(after_train),
         'before_holdout_rows': before['n_rows'].sum(), 'after_holdout_rows': after['n_rows'].sum(),
         'before_train_rows': before_train['n_rows'].sum(), 'after_train_rows': after_train['n_rows'].sum()}
(artifacts / 'fixed-split.json').write_text(json.dumps(split, indent=2)+'\n')
recovery = json.loads((artifacts / 'conversion-resume.json').read_text())
for shard in recovery['retained_shards']:
    path = Path(shard['path'])
    assert hashlib.sha256(path.read_bytes()).hexdigest() == shard['sha256']
    assert path.stat().st_mtime_ns == shard['mtime_ns']
recovery.update(status='complete after identical-command resume; retained shards unchanged',
                final_replay_checkpoints=len(list(Path('data/metamon_gen9ou_r1500_order_blind_decisions/progress').glob('*.json'))))
(artifacts / 'conversion-resume.json').write_text(json.dumps(recovery, indent=2)+'\n')
for name, manifest, raw in [
    ('rows-before-extended', 'runs/order_blind/leaky_split/holdout_manifest.parquet', True),
    ('rows-after', 'runs/blueprint/split/holdout_manifest.parquet', False),
]:
    run(['fog.eval.replay_slot_order_check', '--config', 'configs/replay_blueprint_cpu.yaml',
         '--manifest', manifest, '--out', str(artifacts / f'{name}.json'),
         *(['--raw', 'data/metamon_gen9ou_r1500'] if raw else [])],
        f'runs/order_blind/{name}-output.txt')
while True:
    r0 = json.loads((artifacts / 'R0-execution.json').read_text())
    if any(r['exit_code'] for r in r0):
        raise SystemExit('R0 failed; do not train the comparison.')
    if r0[-1]['output'] == 'runs/order_blind/R0-holdout-output.txt':
        break
    time.sleep(5)
shutil.copyfile(root / 'runs/order_blind/R0/metrics.csv', artifacts / 'R0-metrics.csv')
run(['fog.eval.replay_slot_order_check', '--config', 'configs/replay_blueprint_cpu.yaml',
     '--manifest', 'runs/order_blind/leaky_split/holdout_manifest.parquet', '--checkpoint',
     'runs/order_blind/R0/checkpoints/bc_best.pt', '--raw', 'data/metamon_gen9ou_r1500',
     '--out', str(artifacts / 'R0-slot-check-extended.json')],
    'runs/order_blind/R0-slot-check-extended-output.txt')
for name, seed in [('R1', 20261201), ('R1-seed2', 20261202)]:
    run(['fog.training.bc', '--config', 'configs/replay_blueprint_cpu.yaml', '--set',
         'bc.manifest=runs/blueprint/split/train_1000.parquet',
         'bc.holdout_manifest=runs/blueprint/split/holdout_manifest.parquet',
         'bc.epochs=8', 'bc.early_stopping_patience=0', 'bc.resume=true',
         'bc.selection_metric=action_ce', f'log.dir=runs/order_blind/{name}', f'seed={seed}'],
        f'runs/order_blind/{name}-output.txt')
    shutil.copyfile(root / f'runs/order_blind/{name}/metrics.csv', artifacts / f'{name}-metrics.csv')
    run(['fog.eval.replay_slot_order_check', '--config', 'configs/replay_blueprint_cpu.yaml',
         '--manifest', 'runs/blueprint/split/holdout_manifest.parquet', '--checkpoint',
         f'runs/order_blind/{name}/checkpoints/bc_best.pt', '--out',
         str(artifacts / f'{name}-slot-check.json')], f'runs/order_blind/{name}-slot-check-output.txt')
    run(['fog.eval.replay_holdout_report', '--config', 'configs/replay_blueprint_cpu.yaml',
         '--manifest', 'runs/blueprint/split/holdout_manifest.parquet', '--checkpoint',
         f'runs/order_blind/{name}/checkpoints/bc_best.pt', '--out',
         str(artifacts / f'{name}-holdout.json')], f'runs/order_blind/{name}-holdout-output.txt')
(artifacts / 'stage1-complete.json').write_text(json.dumps({'smoke':True,'status':'complete'})+'\n')
